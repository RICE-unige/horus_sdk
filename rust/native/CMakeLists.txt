cmake_minimum_required(VERSION 3.16)
project(horus_native_ros LANGUAGES CXX)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosidl_typesupport_introspection_cpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(map_msgs QUIET)
find_package(nav2_msgs QUIET)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(nvblox_msgs QUIET)
if(NOT TARGET horus_native_core)
  add_subdirectory(core)
endif()
find_package(nlohmann_json 3.9 REQUIRED)
find_package(Threads REQUIRED)

add_library(horus_native_ros SHARED registry.cpp dashboard.cpp bridge_process.cpp mesh_transport.cpp mesh_source_runtime.cpp teach_runtime.cpp costmap_runtime.cpp model_delivery.cpp description_runtime.cpp)
target_sources(horus_native_ros PRIVATE remote_runtime.cpp)
target_sources(horus_native_ros PRIVATE tf_runtime.cpp)
target_sources(horus_native_ros PRIVATE collision_runtime.cpp)
target_link_libraries(horus_native_ros PRIVATE ${tf2_msgs_TARGETS})
target_compile_features(horus_native_ros PRIVATE cxx_std_17)
target_link_libraries(horus_native_ros PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
target_link_libraries(horus_native_ros PRIVATE horus_native_core)
ament_target_dependencies(horus_native_ros PUBLIC rclcpp std_msgs rcl_interfaces rosbag2_cpp rosidl_typesupport_introspection_cpp)
target_link_libraries(horus_native_ros PRIVATE ${sensor_msgs_TARGETS})
target_link_libraries(horus_native_ros PRIVATE ${visualization_msgs_TARGETS} tf2_ros::tf2_ros)
if(nvblox_msgs_FOUND)
  target_compile_definitions(horus_native_ros PRIVATE HORUS_NATIVE_NVBLOX)
  target_link_libraries(horus_native_ros PRIVATE ${nvblox_msgs_TARGETS})
endif()
if(map_msgs_FOUND AND nav2_msgs_FOUND)
  target_compile_definitions(horus_native_ros PRIVATE HORUS_NATIVE_COSTMAP)
  target_link_libraries(horus_native_ros PRIVATE ${nav_msgs_TARGETS} ${map_msgs_TARGETS} ${nav2_msgs_TARGETS})
endif()
target_include_directories(horus_native_ros PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:include/horus/native>)
set_target_properties(horus_native_ros PROPERTIES VERSION 1.0.0 SOVERSION 1)
install(TARGETS horus_native_ros EXPORT horus_native_rosTargets LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(EXPORT horus_native_rosTargets FILE horus_native_rosTargets.cmake
        NAMESPACE horus_native:: DESTINATION lib/cmake/horus_native_ros)
install(FILES registry.h mesh_transport.h mesh_source_runtime.h teach_runtime.h costmap_runtime.h description_runtime.h remote_runtime.h tf_runtime.h DESTINATION include/horus/native)
install(FILES collision_runtime.h DESTINATION include/horus/native)
