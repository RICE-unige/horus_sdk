cmake_minimum_required(VERSION 3.24)
project(horus_cuda_video LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec>=60 libavutil>=58)
add_library(horus_cuda_video SHARED cuda_video.cu)
set_target_properties(horus_cuda_video PROPERTIES CUDA_STANDARD 17 CXX_STANDARD 17)
target_link_libraries(horus_cuda_video PRIVATE CUDA::cudart PkgConfig::AV)
install(TARGETS horus_cuda_video LIBRARY DESTINATION lib)
